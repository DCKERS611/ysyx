### 本地版本另外需要处理的点

## 三、建议的修复顺序与关键代码

以下片段说明修改点；它们不是已经应用到 WSL 文件的补丁。若决定修改，需同步处理声明、实现与测试。

### 1. 先堵住 `Object_new` 的直接崩溃路径

在 `object.c` 中，分配、写入、复制描述文字的顺序建议改成：

```c
if (size < sizeof(Object) || description == NULL) {
    return NULL;
}

Object *el = calloc(1, size);
if (el == NULL) {
    return NULL;
}

*el = proto;  /* 先按原逻辑填好默认函数指针 */
el->description = strdup(description);
if (el->description == NULL) {
    free(el);
    return NULL;
}
```

这段顺序中的重点是：**任何指针在第一次解引用前必须已经验证**。它只覆盖了创建早期的失败；`init` 过程中创建了别的对象后失败，还需要该类型自己的清理路径。

### 2. 为 `Map_init` 和 `Map_destroy` 定义唯一所有者

建议规定：`Map` 拥有四个房间和一个怪物；方向指针只负责连接房间，不拥有房间；`Room.bad_guy` 指向由 `Map` 管理的怪物。这样可避免沿双向房间连接重复释放。

下面的完整代码采用“`Map` 自己记录所有权”的方案：`calloc` 先将 `rooms[]` 和 `monster` 置空；每创建一个对象就存入 `Map`，失败时返回 0，由 `Object_new` 调用 `Map_destroy` 清理已创建对象。**全部成功以后**再连接方向并设置 `map->start`、`map->location`。正常退出时也由 `Map_destroy` 释放怪物、每个房间各一次，最后释放 `Map` 本身；`main` 在输入循环结束后调用它。

`assert` 可用于指出调用者违反约定，例如传入了本不可能为 `NULL` 的 `self`。对 `calloc` / `strdup` 的失败，应保留可执行的错误处理和清理代码；单独依赖 `assert` 不够，因为定义 `NDEBUG` 时断言会被移除。

### 3. 输入按 `int` 与完整一行处理

可把 `process_input` 开头的输入处理改为以下形式，再使用 `switch(ch)`：

```c
int ch = getchar();
if (ch == EOF) {
    return 0;
}

int rest;
while ((rest = getchar()) != '\n' && rest != EOF) {
    /* 丢弃这一行剩余字符 */
}
```

这样保留了 `getchar()` 的 EOF 信息，也不会把第二个字符误当“必然存在的回车”。如果提示符 `> ` 在非交互输出环境中未及时显示，可在读取前加 `fflush(stdout)`。

## 五、这次审计的学习结论

这个练习中，`proto` 和函数指针完成了手动分派；真正容易使程序崩溃的是**对象创建失败后继续解引用**，容易使程序长期出问题的是**所有权与销毁路径不完整**。检查函数时，按“输入前置条件 → 每次函数调用的返回值 → 下一次解引用 → 正常和错误退出时的资源归属”逐行追踪，比机械地给每个 `if` 添加 `else` 更能发现实际缺陷。

### object.c

```c
#include "object.h"

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int Object_init(void *self)
{
    (void)self;
    return 1;
}

void Object_describe(void *self)
{
    Object *obj = self;
    if (obj == NULL || obj->description == NULL) return;
    printf("%s.\n", obj->description);
}

void Object_destroy(void *self)
{
    Object *obj = self;
    if (obj == NULL) return;
    free(obj->description);
    free(obj);
}

void *Object_move(void *self, Direction direction)
{
    (void)self;
    (void)direction;
    puts("You can't go that direction.");
    return NULL;
}

int Object_attack(void *self, int damage)
{
    (void)self;
    (void)damage;
    puts("You can't attack that.");
    return 0;
}

void *Object_new(size_t size, Object proto, const char *description)
{
    if (size < sizeof(Object) || description == NULL) return NULL;

    if (proto.init == NULL) proto.init = Object_init;
    if (proto.describe == NULL) proto.describe = Object_describe;
    if (proto.destroy == NULL) proto.destroy = Object_destroy;
    if (proto.move == NULL) proto.move = Object_move;
    if (proto.attack == NULL) proto.attack = Object_attack;

    Object *el = calloc(1, size);
    if (el == NULL) return NULL;

    *el = proto;
    el->description = strdup(description);
    if (el->description == NULL) {
        free(el);
        return NULL;
    }

    if (!el->init(el)) {
        el->destroy(el);
        return NULL;
    }

    return el;
}
```

### ex19.c

```c
#include "ex19.h"

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int Monster_init(void *self)
{
    Monster *monster = self;
    if (monster == NULL) return 0;
    monster->hit_points = 10;
    return 1;
}

int Monster_attack(void *self, int damage)
{
    Monster *monster = self;
    if (monster == NULL || damage < 0) return 0;

    if (monster->hit_points == 0) {
        puts("The monster is already dead.");
        return 1;
    }

    if (damage >= monster->hit_points) {
        monster->hit_points = 0;
    } else {
        monster->hit_points -= damage;
    }

    printf("You hit the monster for %d damage.\n", damage);
    if (monster->hit_points == 0) {
        puts("The monster is dead!");
        return 1;
    }

    printf("The monster has %d hit points left.\n", monster->hit_points);
    return 0;
}

Object MonsterProto = {
    .init = Monster_init,
    .attack = Monster_attack
};

void *Room_move(void *self, Direction direction)
{
    Room *room = self;
    if (room == NULL) return NULL;

    Room *next = NULL;
    switch (direction) {
        case NORTH: next = room->north; break;
        case SOUTH: next = room->south; break;
        case EAST:  next = room->east;  break;
        case WEST:  next = room->west;  break;
        default: return NULL;
    }

    if (next == NULL) {
        puts("You can't go that direction.");
        return NULL;
    }

    next->proto.describe(next);
    return next;
}

int Room_attack(void *self, int damage)
{
    Room *room = self;
    if (room == NULL || damage < 0) return 0;
    if (room->bad_guy == NULL) {
        puts("There's nothing to attack here.");
        return 0;
    }
    return room->bad_guy->proto.attack(room->bad_guy, damage);
}

Object RoomProto = {
    .move = Room_move,
    .attack = Room_attack
};

void *Map_move(void *self, Direction direction)
{
    Map *map = self;
    if (map == NULL || map->location == NULL) return NULL;

    Room *next = map->location->proto.move(map->location, direction);
    if (next != NULL) map->location = next;
    return next;
}

int Map_attack(void *self, int damage)
{
    Map *map = self;
    if (map == NULL || map->location == NULL) return 0;
    return map->location->proto.attack(map->location, damage);
}

void Map_destroy(void *self)
{
    Map *map = self;
    if (map == NULL) return;

    if (map->monster != NULL) {
        map->monster->proto.destroy(map->monster);
    }
    for (size_t i = 0; i < ROOM_COUNT; ++i) {
        if (map->rooms[i] != NULL) {
            map->rooms[i]->proto.destroy(map->rooms[i]);
        }
    }
    Object_destroy(map);
}

int Map_init(void *self)
{
    Map *map = self;
    if (map == NULL) return 0;

    /* calloc 已把所有拥有指针置为 NULL；失败时交给 Map_destroy 清理。 */
    map->rooms[0] = NEW(Room, "The great hall");
    if (map->rooms[0] == NULL) return 0;
    map->rooms[1] = NEW(Room, "The throne room");
    if (map->rooms[1] == NULL) return 0;
    map->rooms[2] = NEW(Room, "The arena");
    if (map->rooms[2] == NULL) return 0;
    map->rooms[3] = NEW(Room, "The kitchen");
    if (map->rooms[3] == NULL) return 0;
    map->monster = NEW(Monster, "The evil minotaur");
    if (map->monster == NULL) return 0;

    Room *hall = map->rooms[0];
    Room *throne = map->rooms[1];
    Room *arena = map->rooms[2];
    Room *kitchen = map->rooms[3];

    hall->north = throne;
    throne->south = hall;
    throne->west = arena;
    arena->east = throne;
    throne->east = kitchen;
    kitchen->west = throne;
    arena->bad_guy = map->monster;

    map->start = hall;
    map->location = hall;
    return 1;
}

Object MapProto = {
    .init = Map_init,
    .destroy = Map_destroy,
    .move = Map_move,
    .attack = Map_attack
};

static int process_input(Map *game)
{
    printf("> ");
    fflush(stdout);

    int ch = getchar();
    if (ch == EOF) return 0;

    int rest;
    while ((rest = getchar()) != '\n' && rest != EOF) {
        /* 一次只处理一条命令，其余字符丢弃。 */
    }

    switch (ch) {
        case 'n': game->proto.move(game, NORTH); break;
        case 's': game->proto.move(game, SOUTH); break;
        case 'e': game->proto.move(game, EAST);  break;
        case 'w': game->proto.move(game, WEST);  break;
        case 'a':
            if (game->proto.attack(game, rand() % 4)) return 0;
            break;
        case 'l':
            game->location->proto.describe(game->location);
            break;
        case 'q':
            return 0;
        default:
            puts("Unknown command.");
    }

    return 1;
}

int main(void)
{
    srand((unsigned)time(NULL));

    Map *game = NEW(Map, "The Hall of the Minotaur");
    if (game == NULL) {
        fputs("Failed to create the game map.\n", stderr);
        return EXIT_FAILURE;
    }

    game->location->proto.describe(game->location);
    while (process_input(game)) {
        /* 直到 q、EOF 或击败怪物 */
    }

    game->proto.destroy(game);
    return EXIT_SUCCESS;
}
```

```makefile
	$(CC) $(CFLAGS) ex1.c -o ex1

	$(CC) $(EX19_CFLAGS) -c ex19.c -o ex19.o

	$(CC) $(EX19_CFLAGS) -c object.c -o object.o
```

这份代码还保留了教学示例的简单边界：随机伤害可能为 0，`q`、EOF、击败怪物都会正常清理并退出；它不把 `assert` 当作内存分配失败的防护。要练习更深的故障注入，可在独立副本中让第 N 次分配失败，并观察 `Map_destroy` 是否只释放已创建对象。
