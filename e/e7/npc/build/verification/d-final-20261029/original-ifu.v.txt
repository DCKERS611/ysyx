module ifu #(parameter RESET_PC = 32'h8000_0000) (
    input clk,
    input rst,
    input [31:0] next_pc,
    input [31:0] ifu_rdata,
    input is_load,
    input is_store,
    input ifu_respValid,
    input lsu_respValid,

    output [31:0] pc,
    output reg [31:0] inst,
    output [31:0] ifu_raddr,
    output inst_valid,
    output commit,
    output load_req,
    output ifu_reqValid,
    output lsu_reqValid
);

    // 置 1 开启随机请求等待，置 0 在可发送时立即发请求
    localparam RANDOM_REQ = 1'b0;

    localparam IDLE = 2'b00;
    localparam WAIT = 2'b01;
    localparam MEM_WAIT = 2'b10;

    wire [1:0] state; 
    reg  [1:0] next_state;
    
    always @(*) begin
        next_state = state;
        case (state)
            IDLE: begin
                if (ifu_reqValid)
                    next_state = WAIT;
            end

            WAIT: begin
                if (ifu_respValid) 
                    next_state = is_mem ? MEM_WAIT : IDLE;
            end

            MEM_WAIT: begin
                if (lsu_sent && lsu_respValid) 
                    next_state = IDLE;
            end

            default: begin
                next_state = IDLE;
            end
        endcase
    end

    reg [7:0] req_lfsr;
    reg lsu_sent;

    // 已发送标志只阻止重复请求，不增加状态或固定等待拍
    always @(posedge clk) begin
        if (rst)
            lsu_sent <= 1'b0;
        else if (commit)
            lsu_sent <= 1'b0;
        else if (lsu_reqValid)
            lsu_sent <= 1'b1;
    end

    always @(posedge clk) begin
        if (rst)
            req_lfsr <= 8'h01;
        else 
            req_lfsr <= {
                req_lfsr[6:0],
                req_lfsr[7]^req_lfsr[5]^
                        req_lfsr[4]^req_lfsr[3]
            };
    end
	
    assign ifu_raddr = pc;

    // IDLE时才发出取指请求; 收到取指resp/等待load数据时,指令才有效
    // assign ifu_reqValid = !rst && (state == IDLE);
    assign ifu_reqValid = !rst && (state == IDLE) && (!RANDOM_REQ || req_lfsr[0]);
    // 收到取指回复后指令有效，访存等待期间保留译码信息
    assign inst_valid = !rst &&
        (((state == WAIT) && ifu_respValid) || (state == MEM_WAIT));

    wire normal_done;   // 非load指令   && 收到resp就能提交
    wire mem_done;     // load指令
    wire is_mem;

    assign is_mem = is_load || is_store;
    assign normal_done = (state == WAIT) && !is_mem && ifu_respValid;
    assign mem_done = (state == MEM_WAIT) && lsu_sent && lsu_respValid;
    assign commit = !rst && (normal_done || mem_done);
    // 可在取指回复拍直接请求；若随机条件禁止，则在 MEM_WAIT 中重试
    // 发出后由 lsu_sent 禁止重复请求，直到本条指令完成
    assign lsu_reqValid = !rst && !lsu_sent && !lsu_respValid &&
        (((state == WAIT) && ifu_respValid && is_mem) || (state == MEM_WAIT)) &&
        (!RANDOM_REQ || req_lfsr[1]);
    assign load_req = is_load && lsu_reqValid; 

    Reg #(2, IDLE) u_state (
        .clk(clk),
        .rst(rst),
        .din(next_state),
        .dout(state),
        .wen(1'b1)
    );

    Reg #(32, RESET_PC) u_pc (
        .clk(clk),
        .rst(rst),
        .din(next_pc),
        .dout(pc),
        .wen(commit)
    );
    
    always @(*) begin
        if (rst)
            inst = 32'b0;
        else
            inst = ifu_rdata;
    end
endmodule
